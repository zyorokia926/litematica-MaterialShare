package dev.materialshare.client;

import fi.dy.masa.litematica.data.DataManager;
import fi.dy.masa.litematica.schematic.LitematicaSchematic;
import fi.dy.masa.litematica.schematic.placement.SchematicPlacement;
import fi.dy.masa.litematica.selection.AreaSelection;
import fi.dy.masa.litematica.selection.Box;
import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.function.BooleanSupplier;
import java.util.function.Supplier;
import net.minecraft.client.MinecraftClient;
import net.minecraft.util.math.BlockPos;
import org.lwjgl.glfw.GLFW;

/**
 * TEMPORARY, never committed to src: the steps of a director for the introduction video, done one after the other on the
 * game ticks. A director (ShootDirector) only lists steps: commands, waits, the fake mouse on texts found by their words,
 * the start and end of a clip, marks for the video editor, and cues that the two players wait for (files in the shots folder,
 * both games run on the same PC).
 */
public final class TmpShot {
    interface Step {
        /** true when the step is done */
        boolean run(MinecraftClient c);
    }

    static final Deque<Step> STEPS = new ArrayDeque<>();
    static final Path OUT = Path.of(System.getProperty("shoot.out", "shots"));
    static int waited;
    static boolean failed;

    static MinecraftClient mc() {
        return MinecraftClient.getInstance();
    }

    /** one tick: runs the current step, and the next ones while they finish at once */
    public static void tick(MinecraftClient c) {
        for (int guard = 0; guard < 50 && !STEPS.isEmpty() && !failed; guard++) {
            Step s = STEPS.peekFirst();
            boolean done;
            try {
                done = s.run(c);
            } catch (Exception e) {
                TmpProbe.out("EXC " + e);
                e.printStackTrace(System.out);
                failed = true;
                abort();
                TmpRecorder.stop();
                c.scheduleStop();
                return;
            }
            if (!done) {
                waited++;
                return;
            }
            STEPS.pollFirst();
            waited = 0;
        }
    }

    public static boolean idle() {
        return STEPS.isEmpty();
    }

    // ---- the steps ----

    public static void then(Runnable r) {
        STEPS.add(c -> {
            r.run();
            return true;
        });
    }

    public static void ticks(int n) {
        STEPS.add(c -> waited >= n);
    }

    public static void seconds(double s) {
        ticks((int) Math.round(s * 20));
    }

    /** waits until the condition is true; fails the shoot after `timeoutTicks` */
    public static void until(String what, BooleanSupplier ok, int timeoutTicks) {
        STEPS.add(c -> {
            if (ok.getAsBoolean()) {
                return true;
            }
            if (waited > timeoutTicks) {
                throw new IllegalStateException("timed out: " + what);
            }
            return false;
        });
    }

    /**
     * a command run by the server console: shoot.py passes the lines of server_cmds.txt to the server. (Commands of a player
     * count as chat: after about ten the server kicks him for spamming.) Absolute coordinates only.
     */
    public static void cmd(String command) {
        then(() -> {
            try {
                Files.writeString(OUT.resolve("server_cmds.txt"), command + "\n", java.nio.charset.StandardCharsets.UTF_8,
                        java.nio.file.StandardOpenOption.CREATE, java.nio.file.StandardOpenOption.APPEND);
            } catch (Exception e) {
                throw new IllegalStateException(e);
            }
        });
    }

    /** waits until the server ran every command so far (shoot.py writes how many) */
    public static void commandsDone() {
        until("server commands", () -> {
            try {
                long sent = Files.readAllLines(OUT.resolve("server_cmds.txt")).size();
                Path done = OUT.resolve("server_cmds_done.txt");
                return Files.exists(done) && Long.parseLong(Files.readString(done).strip()) >= sent;
            } catch (Exception e) {
                return false;
            }
        }, 20 * 60);
        ticks(10);
    }

    public static void say(String text) {
        then(() -> TmpProbe.out(text));
    }

    /** glides the cursor to a text found on the screen (waits until it is drawn), in `millis`, and waits until it arrived */
    public static void point(String what, Supplier<TmpLocate.Spot> spot, long millis) {
        STEPS.add(c -> {
            if (spot.get() != null) {
                return true;
            }
            if (waited > 200) {
                throw new IllegalStateException("not on the screen: " + what + "; the screen shows " + TmpLocate.last.size() + " texts:\n" + TmpLocate.dump());
            }
            return false;
        });
        then(() -> {
            TmpLocate.Spot s = spot.get();
            TmpInput.moveTo(s.cx(), s.cy(), millis);
        });
        until("cursor arrived", () -> !TmpInput.moving(), 400);
    }

    public static void pointAt(String exactWords, long millis) {
        point(exactWords, () -> TmpLocate.exact(exactWords, 0), millis);
    }

    /** glides the cursor to a place on the screen (fractions of the width and height) */
    public static void glide(double fx, double fy, long millis) {
        then(() -> TmpInput.moveTo(mc().getWindow().getScaledWidth() * fx, mc().getWindow().getScaledHeight() * fy, millis));
        until("cursor arrived", () -> !TmpInput.moving(), 400);
    }

    public static void click() {
        then(() -> TmpInput.click(0));
        ticks(2);
    }

    public static void rightClick() {
        then(() -> TmpInput.click(1));
        ticks(2);
    }

    public static void wheel(int notches, int ticksBetween) {
        for (int i = 0; i < Math.abs(notches); i++) {
            then(() -> TmpInput.scroll(notches > 0 ? 1 : -1));
            ticks(ticksBetween);
        }
    }

    public static void type(String text) {
        then(() -> TmpInput.type(text));
    }

    public static void key(int glfwKey) {
        then(() -> TmpInput.key(glfwKey));
    }

    public static void hold(int glfwKey) {
        then(() -> TmpInput.hold(glfwKey));
    }

    public static void release(int glfwKey) {
        then(() -> TmpInput.release(glfwKey));
    }

    /** starts a clip <lang>/<name>.mp4 (and .jsonl) */
    public static void rec(String name) {
        then(() -> TmpRecorder.start(OUT.resolve(name).toString().replace('\\', '/')));
        ticks(2);
    }

    public static void cut() {
        then(TmpRecorder::stop);
    }

    /** a mark in the log of the clip: where a caption, a zoom or a cut of the video editor belongs */
    public static void mark(String name) {
        then(() -> TmpRecorder.event("\"ev\":\"mark\",\"name\":\"" + name + "\""));
    }

    /** a cue for the other player */
    public static void signal(String name) {
        then(() -> {
            try {
                Files.writeString(OUT.resolve("cue_" + name), "1");
            } catch (Exception e) {
                throw new IllegalStateException(e);
            }
        });
    }

    public static void await(String name) {
        until("cue " + name, () -> {
            if (Files.exists(OUT.resolve("cue_abort"))) {
                throw new IllegalStateException("the other player stopped");
            }
            return Files.exists(OUT.resolve("cue_" + name));
        }, 20 * 300);
    }

    /** tells the other player that this one failed */
    static void abort() {
        try {
            Files.writeString(OUT.resolve("cue_abort"), "1");
        } catch (Exception ignored) {
        }
    }

    // ---- the setting of the shoot ----

    /** the window without a frame and exactly the size of the video; no chat on the screen */
    public static void cleanWindow(int w, int h) {
        then(() -> {
            long win = mc().getWindow().getHandle();
            GLFW.glfwSetWindowAttrib(win, GLFW.GLFW_DECORATED, GLFW.GLFW_FALSE);
            GLFW.glfwSetWindowSize(win, w, h);
        });
        ticks(10);
        then(() -> TmpProbe.out("window " + mc().getWindow().getFramebufferWidth() + "x" + mc().getWindow().getFramebufferHeight()
                + " gui scale " + mc().getWindow().getScaleFactor()));
    }

    public static void clearChat() {
        then(() -> mc().inGameHud.getChatHud().clear(false));
    }

    /** a schematic of the blocks between a and b (as the client sees them), saved and placed at a; it becomes the selected placement */
    public static void schematicOf(BlockPos a, BlockPos b, String name) {
        then(() -> {
            AreaSelection area = new AreaSelection();
            area.setName(name);
            area.addSubRegionBox(new Box(a, b, name), false);
            LitematicaSchematic s = LitematicaSchematic.createFromWorld(mc().world, area, new LitematicaSchematic.SchematicSaveInfo(false, false), "MaterialShare", m -> TmpProbe.out("litematica: " + m));
            File dir = new File(mc().runDirectory, "schematics");
            dir.mkdirs();
            s.writeToFile(dir, name, true);
            var manager = DataManager.getSchematicPlacementManager();
            for (var old : java.util.List.copyOf(manager.getAllSchematicsPlacements())) {
                manager.removeSchematicPlacement(old);
            }
            SchematicPlacement p = SchematicPlacement.createFor(s, new BlockPos(Math.min(a.getX(), b.getX()), Math.min(a.getY(), b.getY()), Math.min(a.getZ(), b.getZ())), name, true, true);
            manager.addSchematicPlacement(p, false);
            manager.setSelectedSchematicPlacement(p);
            TmpProbe.out("schematic " + name + " placed");
        });
    }
}
