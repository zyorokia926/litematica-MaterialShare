    /**
     * a cut where both players are filmed (split screen): P2 records while P1 acts; P1's steps are `alice`. Clips: <clip>_p1 and
     * <clip>_p2. `lead` = seconds that both films run before the first action.
     */
    static void split(String clip, double lead, Runnable alice) {
        if (P1) {
            await("r_" + clip);
            signal("g_" + clip);
            rec(clip + "_p1");
            seconds(lead);
            alice.run();
            signal("e_" + clip);
            seconds(0.3);
            cut();
        } else {
            signal("r_" + clip);
            await("g_" + clip);
            rec(clip + "_p2");
            await("e_" + clip);
            seconds(0.3);
            cut();
        }
    }

    /** a cut that only P1 films (P2 goes on at once to its next cut and waits there) */
    static void solo(String clip, double lead, Runnable alice) {
        if (P1) {
            rec(clip);
            seconds(lead);
            alice.run();
            cut();
        }
    }

    /** the text of a language key without its numbers and brackets ("余りの共有 (%s)" -> "余りの共有"), to find a button by it */
    static String lab(String key) {
        String s = Lang.t(key, "\u0001");
        int i = s.indexOf('\u0001');
        return (i >= 0 ? s.substring(0, i) : s).replaceAll("[\\s(（]+$", "");
    }

    // the storage hut: x 12..16, z 0..3 (offsets from the house corner), the ground block of corner A and the post of corner B
    static String at(int dx, int dy, int dz) {
        return (o.getX() + dx) + " " + (o.getY() + dy) + " " + (o.getZ() + dz);
    }

    static BlockPos hut(int dx, int dy, int dz) {
        return o.add(dx, dy, dz);
    }

    /** the player stands at (px, py, pz) and looks at (fx, fy, fz) (offsets from the house corner); the angles come from the eyes ("tp ... facing" looks from the feet: pitch 0) */
    static void face(String player, double px, double py, double pz, double fx, double fy, double fz) {
        double dx = fx - px, dz = fz - pz, dy = fy - (py + 1.62);
        double yaw = Math.toDegrees(Math.atan2(-dx, dz));
        double pitch = -Math.toDegrees(Math.atan2(dy, Math.sqrt(dx * dx + dz * dz)));
        cmd(String.format(java.util.Locale.ROOT, "tp %s %.2f %.2f %.2f %.1f %.1f", player, o.getX() + px, o.getY() + py, o.getZ() + pz, yaw, pitch));
    }

    static void twoPartSchematic() {
        then(() -> {
            String name = "MaterialShare Demo House";
            String first = JA ? "1階" : "Floor 1", roof = JA ? "屋根" : "Roof";
            var area = new fi.dy.masa.litematica.selection.AreaSelection();
            area.setName(name);
            area.addSubRegionBox(new fi.dy.masa.litematica.selection.Box(o.add(-1, 0, -1), o.add(9, 3, 7), first), false);
            area.addSubRegionBox(new fi.dy.masa.litematica.selection.Box(o.add(-1, 4, -1), o.add(9, 7, 7), roof), false);
            var mc = MinecraftClient.getInstance();
            var s = fi.dy.masa.litematica.schematic.LitematicaSchematic.createFromWorld(mc.world, area,
                    new fi.dy.masa.litematica.schematic.LitematicaSchematic.SchematicSaveInfo(false, false), "MaterialShare", m -> out("litematica: " + m));
            java.io.File dir = new java.io.File(mc.runDirectory, "schematics");
            dir.mkdirs();
            s.writeToFile(dir, name, true);
            // read back from the file: Syncmatica sends the schematic's file, a schematic made in memory has none
            s = fi.dy.masa.litematica.schematic.LitematicaSchematic.createFromFile(dir, name + ".litematic");
            fi.dy.masa.litematica.data.SchematicHolder.getInstance().addSchematic(s, true);
            var manager = DataManager.getSchematicPlacementManager();
            for (var old : java.util.List.copyOf(manager.getAllSchematicsPlacements())) {
                manager.removeSchematicPlacement(old);
            }
            var p = fi.dy.masa.litematica.schematic.placement.SchematicPlacement.createFor(s, o.add(-1, 0, -1), name, true, true);
            manager.addSchematicPlacement(p, false);
            manager.setSelectedSchematicPlacement(p);
            out("schematic placed: " + name + " (" + first + ", " + roof + ")");
        });
    }

    public static void init() {
        ClientTickEvents.END_CLIENT_TICK.register(c -> {
            tick++;
            if (c.world == null || c.player == null) {
                if (planned) {  // kicked or disconnected during the shoot: a clip would miss its start, so stop
                    out("FAIL disconnected during the shoot");
                    TmpShot.abort();
                    TmpRecorder.stop();
                    c.scheduleStop();
                    planned = false;
                    return;
                }
                if ((!connecting || tick - tried > 400) && tick > 200 && c.currentScreen != null
                        && (c.currentScreen instanceof TitleScreen || c.currentScreen.getClass().getSimpleName().contains("Disconnected"))) {
                    connecting = true;
                    tried = tick;
                    ConnectScreen.connect(c.currentScreen, c, ServerAddress.parse("127.0.0.1:" + PORT), new ServerInfo("t", "127.0.0.1:" + PORT, false));
                }
                return;
            }
            if (c.player.age < 250) {
                return;
            }
            if (!planned) {
                planned = true;
                JA = "ja_jp".equals(c.options.language);
                dev.materialshare.client.config.Configs.Generic.SHOW_INTRO.setBooleanValue(false);
                dev.materialshare.client.config.Configs.Generic.AUTO_IMPORT.setBooleanValue(false);
                // the house stands a little away from the spawn; both players use the same corner
                BlockPos spawn = c.world.getSpawnPos();
                o = new BlockPos(spawn.getX() + 6, c.world.getTopY(net.minecraft.world.Heightmap.Type.MOTION_BLOCKING, spawn.getX(), spawn.getZ()), spawn.getZ() - 4);
                out("house corner " + o);
                plan();
            }
            TmpShot.tick(c);
        });
    }
