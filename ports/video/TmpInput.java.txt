package dev.materialshare.client;

import java.lang.reflect.Method;
import java.util.HashSet;
import java.util.Set;
import net.minecraft.client.MinecraftClient;
import net.minecraft.client.Mouse;
import org.lwjgl.glfw.GLFW;

/**
 * TEMPORARY, never committed to src: a fake mouse and keyboard for the introduction video. It calls the game's own input
 * handlers (as GLFW would), so the screens react as to a person: buttons light up under the cursor, clicks and the wheel
 * work. The cursor glides to its target with an ease (moved every frame by TmpRecorderMixin). Positions are in GUI units
 * (the coordinates of the screens); the log has them in pixels of the video, for the cursor that the video editor draws.
 */
public final class TmpInput {
    static final Set<Integer> HELD = new HashSet<>();  // keys that count as held down (TmpKeysMixin)
    static double x = -1, y = -1, fromX, fromY, toX, toY;
    static long moveStart, moveNanos;
    static boolean moving;
    static Method cursorPos, mouseButton, mouseScroll, charTyped;

    static long window() {
        return MinecraftClient.getInstance().getWindow().getHandle();
    }

    static double scale() {
        return MinecraftClient.getInstance().getWindow().getScaleFactor();
    }

    static Method find(Class<?> c, String name, Class<?>... args) throws Exception {
        Method m = c.getDeclaredMethod(name, args);
        m.setAccessible(true);
        return m;
    }

    static void init() throws Exception {
        if (cursorPos == null) {
            cursorPos = find(Mouse.class, "onCursorPos", long.class, double.class, double.class);
            mouseButton = find(Mouse.class, "onMouseButton", long.class, int.class, int.class, int.class);
            mouseScroll = find(Mouse.class, "onMouseScroll", long.class, double.class, double.class);
            charTyped = find(net.minecraft.client.Keyboard.class, "onChar", long.class, int.class, int.class);
        }
    }

    /** the cursor glides to (gx, gy) (GUI units) in `millis` */
    public static void moveTo(double gx, double gy, long millis) {
        if (x < 0) {
            x = gx;
            y = gy;
        }
        fromX = x;
        fromY = y;
        toX = gx;
        toY = gy;
        moveStart = System.nanoTime();
        moveNanos = Math.max(1, millis) * 1_000_000L;
        moving = true;
    }

    public static boolean moving() {
        return moving;
    }

    /** every frame, before it is drawn */
    public static void frame() {
        if (!moving) {
            return;
        }
        double t = Math.min(1, (System.nanoTime() - moveStart) / (double) moveNanos);
        double e = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;  // ease in and out
        x = fromX + (toX - fromX) * e;
        y = fromY + (toY - fromY) * e;
        put();
        if (t >= 1) {
            moving = false;
        }
    }

    static void put() {
        try {
            init();
            double s = scale();
            cursorPos.invoke(MinecraftClient.getInstance().mouse, window(), x * s, y * s);
            TmpRecorder.event("\"ev\":\"cursor\",\"x\":" + Math.round(x * s) + ",\"y\":" + Math.round(y * s));
        } catch (Exception e) {
            TmpProbe.out("INPUT move failed " + e);
        }
    }

    /** a click of the left (0) or right (1) button where the cursor is */
    public static void click(int button) {
        try {
            init();
            Mouse m = MinecraftClient.getInstance().mouse;
            int mods = (HELD.contains(GLFW.GLFW_KEY_LEFT_SHIFT) ? GLFW.GLFW_MOD_SHIFT : 0) | (HELD.contains(GLFW.GLFW_KEY_LEFT_CONTROL) ? GLFW.GLFW_MOD_CONTROL : 0);
            TmpRecorder.event("\"ev\":\"click\",\"button\":" + button + ",\"mods\":" + mods);
            mouseButton.invoke(m, window(), button, GLFW.GLFW_PRESS, mods);
            mouseButton.invoke(m, window(), button, GLFW.GLFW_RELEASE, mods);
        } catch (Exception e) {
            TmpProbe.out("INPUT click failed " + e);
        }
    }

    /** one notch of the wheel (up = 1, down = -1) */
    public static void scroll(int notches) {
        try {
            init();
            TmpRecorder.event("\"ev\":\"wheel\",\"n\":" + notches);
            mouseScroll.invoke(MinecraftClient.getInstance().mouse, window(), 0.0, (double) notches);
        } catch (Exception e) {
            TmpProbe.out("INPUT wheel failed " + e);
        }
    }

    /** a key press and release (GLFW key code) */
    public static void key(int key) {
        TmpRecorder.event("\"ev\":\"key\",\"key\":" + key);
        var kb = MinecraftClient.getInstance().keyboard;
        int scan = GLFW.glfwGetKeyScancode(key);
        kb.onKey(window(), key, scan, GLFW.GLFW_PRESS, 0);
        kb.onKey(window(), key, scan, GLFW.GLFW_RELEASE, 0);
    }

    /** holds a key (for example GLFW_KEY_LEFT_SHIFT) until release() */
    public static void hold(int key) {
        HELD.add(key);
        TmpRecorder.event("\"ev\":\"hold\",\"key\":" + key);
    }

    public static boolean held(int key) {
        return HELD.contains(key);
    }

    public static void release(int key) {
        HELD.remove(key);
        TmpRecorder.event("\"ev\":\"release\",\"key\":" + key);
    }

    /** types text into the focused text field */
    public static void type(String text) {
        try {
            init();
            TmpRecorder.event("\"ev\":\"type\",\"n\":" + text.length());
            for (int cp : text.codePoints().toArray()) {
                charTyped.invoke(MinecraftClient.getInstance().keyboard, window(), cp, 0);
            }
        } catch (Exception e) {
            TmpProbe.out("INPUT type failed " + e);
        }
    }
}
